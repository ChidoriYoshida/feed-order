var SHEET_NAME = 'シート1'; function doGet(e) { var params = e.parameter; var callback = params.callback || 'callback'; var action = params.action || 'get'; if (action === 'post') { try { var data = JSON.parse(decodeURIComponent(params.data || '{}')); var orders = data.orders || []; var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME) || SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); // ヘッダがなければ追加 if (sheet.getLastRow() === 0) { sheet.appendRow(['ID','配送日','担当者','農場','タンク','銘柄','数量(t)','添加剤','紙袋','受信日時']); } orders.forEach(function(o) { sheet.appendRow([ o.id || '', o.date || '', o.staff || '', o.farm || '', o.tank || '', o.brand || '', o.qty || 0, o.additive === 'yes' ? '有' : '無', o.isBag ? '紙袋' : 'タンク', Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy/MM/dd HH:mm:ss') ]); }); var result = JSON.stringify({ result: 'success', count: orders.length }); return ContentService.createTextOutput(callback + '(' + result + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } catch(err) { var errResult = JSON.stringify({ result: 'error', message: err.message }); return ContentService.createTextOutput(callback + '(' + errResult + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } } // action === 'get' var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME) || SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var rows = sheet.getDataRange().getValues(); if (rows.length <= 1) { return ContentService.createTextOutput(callback + '({"orders":[]})') .setMimeType(ContentService.MimeType.JAVASCRIPT); } var headers = rows[0]; var orders = rows.slice(1).map(function(row) { var o = {}; headers.forEach(function(h, i) { o[h] = row[i]; }); return o; }); var result = JSON.stringify({ orders: orders }); return ContentService.createTextOutput(callback + '(' + result + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); }
